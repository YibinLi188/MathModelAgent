\documentclass[fontset=none, 12pt, a4paper]{ctexart}

% ---------- 页面：A4，上下左右 2.5cm ----------
\usepackage[a4paper, top=2.5cm, bottom=2.5cm, left=2.5cm, right=2.5cm]{geometry}
\usepackage{xcolor}

% ---------- 数学 ----------
\usepackage{amsmath}

% ---------- 字体（fontspec + ctex，xelatex 编译） ----------
% Windows 使用常见宋/黑体；Docker/Linux 使用 Noto CJK，避免自动猜测后挂起。
\IfFontExistsTF{SimSun}{%
  \setCJKmainfont{SimSun}
  \setCJKsansfont{SimHei}
  \newCJKfontfamily\paperheadingfont{SimHei}
}{%
  \setCJKmainfont{Noto Serif CJK SC}
  \setCJKsansfont{Noto Sans CJK SC}
  \newCJKfontfamily\paperheadingfont{Noto Sans CJK SC}
}

% ---------- 正文：12.05pt，首行缩进 2em，两端对齐，行距 0.72em ----------
% 12pt 字号近似 12.05pt；0.72em 行距 => 行高 = 1.72 * 字号，\linespread 因子 ≈ 1.43
\linespread{1.43}
\setlength{\parindent}{2em}

% ---------- 标题编号（\ctexset 配置） ----------
% 一级标题用中文数字“一、二、三、”，二级及以下用 1.1 / 1.1.1
\ctexset{
  section/number       = \chinese{section},
  subsection/number    = \arabic{section}.\arabic{subsection},
  subsubsection/number = \arabic{section}.\arabic{subsection}.\arabic{subsubsection},
}

  {\color{black}\fontsize{12.05pt}{14.5pt}\paperheadingfont\bfseries}
  {\arabic{section}.\arabic{subsection}.\arabic{subsubsection}}{1em}{}

\titlespacing{\subsection}    {0pt}{1.15em}{0.55em}
\titlespacing{\subsubsection} {0pt}{1.15em}{0.55em}
\newcommand{\paperneedlines}[1]{%
  \par\begingroup
  \dimen0=\pagegoal\advance\dimen0 by -\pagetotal
  \dimen2=#1\baselineskip
  \ifdim\dimen0<\dimen2\pagebreak[4]\fi
  \endgroup
}
\pretocmd{\section}{\paperneedlines{8}}{}{}
\pretocmd{\subsection}{\paperneedlines{5}}{}{}
\pretocmd{\subsubsection}{\paperneedlines{4}}{}{}

% ---------- 自定义命令（对应 Typst 函数） ----------
% \papertitle{标题}：居中 17.3pt 加粗标题
\newcommand{\papertitle}[1]{%
  \begin{center}
    {\color{black}\fontsize{17.3pt}{22pt}\paperheadingfont\bfseries #1}
  \end{center}
  \vspace{1em}
}
% \abstractcn{摘要正文}{关键词}：摘要后直接进入正文，不制造独立空白页。
\newcommand{\abstractcn}[2]{%
  \begin{center}{\color{black}\fontsize{14pt}{17pt}\paperheadingfont\bfseries 摘要}\end{center}
  #1\par
  \vspace{1em}
  {\paperheadingfont\bfseries 关键字：}#2\par
}
% \referencescn：参考文献节，标题“参考文献”（不带编号但进目录）
\newcommand{\referencescn}{%
  \addcontentsline{toc}{section}{参考文献}%
  \input{references}%
}
% \appendixcn：附录节，标题“附录 A  核心代码”
\newcommand{\appendixcn}{%
  \section*{附录 A\quad 核心代码}%
  \addcontentsline{toc}{section}{附录 A\quad 核心代码}%
  \input{sections/A_code}%
}
% \threelinetable{标题}{列定义}{表头}{表格内容}：三线表，标题在表格上方居中 10.5pt 黑体加粗
% 调用示例：\threelinetable{表 1 示例}{ccc}{A & B & C}{1 & 2 & 3 \\ 4 & 5 & 6}
\newcommand{\threelinetable}[4]{%
  \begin{center}
    {\fontsize{10.5pt}{12.6pt}\paperheadingfont\bfseries #1}\par
    \vspace{0.6em}
    \begin{tabular}{#2}
      \toprule
      #3 \\
      \midrule
      #4 \\
      \bottomrule
    \end{tabular}
  \end{center}
}
